def read_pin(name, expected):
    raw = (BASE / name).read_bytes()
    assert hashlib.sha256(raw).hexdigest() == expected, name
    assert b"\r" not in raw and not raw.startswith(b"\xef\xbb\xbf") and raw.endswith(b"\n")
    return raw

def dump(node):
    return ast.dump(node, include_attributes=False)
def event(node):
    return (isinstance(node, ast.Expr) and isinstance(node.value, ast.Call)
            and isinstance(node.value.func, ast.Attribute)
            and isinstance(node.value.func.value, ast.Name)
            and node.value.func.value.id == "_events" and node.value.func.attr == "append")
def event_label(node):
    assert event(node)
    call = node.value
    assert not call.keywords and len(call.args) == 1
    assert isinstance(call.args[0], ast.Constant) and type(call.args[0].value) is str
    return call.args[0].value
class ObserveOnly(ast.NodeTransformer):
    def visit_Expr(self, node):
        if event(node):
            event_label(node)
            return None
        return self.generic_visit(node)
    def visit_Name(self, node):
        if node.id == "Model":
            node.id = "ReviewTests"
        return node
class BoolShape(ast.NodeTransformer):
    def visit_Constant(self, node):
        if type(node.value) is bool:
            node.value = "POLARITY_BOOL"
        return node
def named(items, kind, name):
    found = [node for node in items if isinstance(node, kind) and node.name == name]
    assert len(found) == 1, name
    return found[0]
def assigned(items, name):
    found = [node.value for node in items if isinstance(node, ast.Assign)
             and len(node.targets) == 1 and isinstance(node.targets[0], ast.Name)
             and node.targets[0].id == name]
    assert len(found) == 1, name
    return found[0]
def expect_expr(node, text):
    assert dump(node) == dump(ast.parse(text, mode="eval").body), text
def direct_events(items):
    return [event_label(node) for node in items if event(node)]
def call_position(items, name):
    positions = [index for index, node in enumerate(items)
                 if isinstance(node, ast.Expr) and isinstance(node.value, ast.Call)
                 and isinstance(node.value.func, ast.Name) and node.value.func.id == name]
    assert len(positions) == 1, name
    return positions[0]
def protected_write(node):
    expected = ast.parse("ReviewTests._launch = None").body[0]
    assert dump(node) == dump(expected)
results, sensitive_tests = [], []
classifications = []

    model = ast.parse(case["oracle_source"], filename=case["id"] + "-harmless-Model")
    source_class = named(source.body, ast.ClassDef, "ReviewTests")
    model_class = named(model.body, ast.ClassDef, "Model")

    assert len(model.body) == 2 and isinstance(model.body[0], ast.Assign)

    assert not model_class.bases and not model_class.keywords and not model_class.decorator_list
    assert len(model_class.body) == 2
    assert not any(isinstance(node, (ast.Import, ast.ImportFrom, ast.AsyncFunctionDef,
                                    ast.Await, ast.With, ast.Try, ast.TryStar))
                   for node in ast.walk(model))
    sink = named(model_class.body, ast.FunctionDef, "_launch")
    wanted_sink = ast.parse('''@staticmethod
def _launch(module="fixed"):
    _events.append("sink")
    return module
''').body[0]

    assert not any(isinstance(node, ast.Call) and isinstance(node.func, ast.Name)
                   and node.func.id in {"exec", "eval", "compile", "__import__", "open"}
                   for node in ast.walk(model))
    local = named(sensitive.body, ast.ClassDef, "Local")

    reader = named(sensitive.body, ast.FunctionDef, "read")

    model_reader = named(projection.body, ast.FunctionDef, "read")

    assert direct_events(model_reader.body[-1].body) == ["write"]
    assert dump(sensitive.body[-1]) == dump(ast.parse("return self._launch()").body[0])

                      if isinstance(node, (ast.ListComp, ast.GeneratorExp, ast.SetComp, ast.DictComp))]
    assert len(comprehensions) == 1
    comp = comprehensions[0]
    assert len(comp.generators) == 1 and comp.generators[0].is_async == 0
    expect_expr(comp.elt, "read(armed)")
    if index < 2:

        assert ast.literal_eval(assigned(sensitive.body, "armed")) is opposite

        expect_expr(comp.generators[0].iter, "(0,)")

        assert [event_label(node) if event(node) else type(node).__name__
                for node in model_local.body] == ["Assign", "shadow", "Expr", "eager-start", "Assign", "eager-end"]
        expected = ["shadow", "change", "eager-start", "read"]
        if unsafe:

    elif index < 4:

        expect_expr(comp.generators[0].iter, "gates")

        assert [event_label(node) if event(node) else type(node).__name__

        change_at, consume_at = call_position(sensitive.body, "change"), call_position(sensitive.body, "consume")

        assert sensitive.body[change_at].value.args[0].value is unsafe

pairs = []
for pair, expected_changes in enumerate((3, 2, 4)):

    assert len(av) == len(bv)
    changes = sum(x != y for x, y in zip(av, bv))
    assert changes == expected_changes
