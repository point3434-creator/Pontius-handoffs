def digest(raw):
    return hashlib.sha256(raw).hexdigest()
def pinned(name, expected):
    raw = (BASE / name).read_bytes()

    return raw

def dump(node):
    return ast.dump(node, include_attributes=False)
def assignment(tree, name):
    return next(n.value for n in tree.body if isinstance(n, ast.Assign)
                and any(isinstance(t, ast.Name) and t.id == name for t in n.targets))
def functions(tree):
    return {n.name: n for n in tree.body if isinstance(n, ast.FunctionDef)}
def summary_fields(tree):
    main = functions(tree)["main"]
    summary = next(n.value for n in ast.walk(main) if isinstance(n, ast.Assign)
                   and any(isinstance(t, ast.Name) and t.id == "summary" for t in n.targets))
    assert isinstance(summary, ast.Dict)
    return {k.value: v for k, v in zip(summary.keys, summary.values, strict=True)}
def has_expression(tree, source, expected_count=1):
    expected = dump(ast.parse(source, mode="eval").body)
    count = sum(dump(n) == expected for n in ast.walk(tree))
    assert count == expected_count, (source, count, expected_count)
def subscript_is(node, root, key):
    return (isinstance(node, ast.Subscript) and isinstance(node.value, ast.Name)
            and node.value.id == root and isinstance(node.slice, ast.Constant)
            and node.slice.value == key)
def domain(tree, root, key, expected):
    values = set()
    for node in ast.walk(tree):
        if isinstance(node, ast.Compare) and any(
                subscript_is(n, root, key) for n in ast.walk(node)):
            for child in ast.walk(node):
                if isinstance(child, ast.Constant) and type(child.value) in {int, float, bool}:
                    assert type(child.value) is int, (root, key, child.value)
                    values.add(child.value)
    assert values == set(expected), (root, key, values, expected)
    return sorted(values)

for role in ("probe", "control"):
    steps = []
    texts = []
    for index, mapping in enumerate(maps, 1):
        old_info, new_info = mapping[role + "_base"], mapping[role + "_new"]
        old_raw = pinned(old_info["path"], old_info["sha256"])
        new_raw = pinned(new_info["path"], new_info["sha256"])
        old, new = old_raw.decode(), new_raw.decode()
        assert old == old.rstrip() + "\n"
        expected = old
        for edit in new_info["edits"]:
            assert expected.count(edit["old"]) == edit["count"]
            expected = expected.replace(edit["old"], edit["new"])
        assert expected == new, (role, index)
        old_tree, new_tree = ast.parse(old), ast.parse(new)
        compile(new, new_info["path"], "exec", dont_inherit=True)
        old_funcs, new_funcs = functions(old_tree), functions(new_tree)
        assert old_funcs.keys() == new_funcs.keys()
        changed = sorted(n for n in old_funcs if dump(old_funcs[n]) != dump(new_funcs[n]))
        assert changed == (["load_pack", "main", "oracle"] if role == "probe" and index == 1
                           else ["load_pack", "main"] if role == "probe"
                           else ["main", "validate_result"]), changed
        for name in ("CAPS", "SLOTS"):
            assert dump(assignment(old_tree, name)) == dump(assignment(new_tree, name))
        if role == "probe":
            assert dump(old_funcs["public_review"]) == dump(new_funcs["public_review"])
        else:
            for name in ("WATCH", "WATCH_SHA", "TIMEOUT", "BASE", "GIT"):
                assert dump(assignment(old_tree, name)) == dump(assignment(new_tree, name))
        diff_raw = "".join(difflib.unified_diff(
            old.splitlines(keepends=True), new.splitlines(keepends=True),
            fromfile=old_info["path"], tofile=new_info["path"])).encode()
        suffix = "v1-from-boundary-v1" if index == 1 else "v2-from-v1"
        diff_name = "class-comprehension-extension-" + role + "-" + suffix + ".diff"
        with (BASE / diff_name).open("xb") as stream:
            stream.write(diff_raw)
        steps.append({"before_sha256": digest(old_raw), "after_sha256": digest(new_raw),
                      "authorized_replacement_replay_equal": True,
                      "changed_functions": changed,
                      "unchanged_functions": sorted(n for n in old_funcs if n not in changed),
                      "diff_path": diff_name, "diff_sha256": digest(diff_raw),
                      "syntax_compiled_not_executed": True})
        texts.append(new)
        trees[(role, index)] = new_tree
    records[role] = steps
