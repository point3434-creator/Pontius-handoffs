def growing_case(h, case):
    cursor, reference = h.empty()
    parents = []
    names = ordinary_names(case["n"])
    for i, name in enumerate(names):
        value = h.tokens.atom("grow-" + str(i))
        h.set(cursor, reference, name, value, i % 4 != 0)
        if parents:
            h.get(parents[-1][0], parents[-1][1], name, "fork_isolation")
        parent = (cursor, reference.fork())
        cursor, reference = h.fork(cursor, reference)
        parents.append(parent)
    assert len(parents) == case["n"]
    h.report["real_cursor_forks"] = len(parents)
    h.ordered(cursor, reference)
    for length in (case["n"] // 4, case["n"] // 2, case["n"]):
        state, expected = parents[length - 1]
        h.ordered(state, expected, prefix="historical", repeats=1)


def ambient(h, n):
    reference = Reference()
    for i, name in enumerate(ordinary_names(n)):
        reference.set(name, h.tokens.atom("ambient-" + str(i)), True)
    version, reference = h.bulk(reference, "construction")
    return h.cursor(version, reference)


def hot_case(h, case):
    cursor, reference = ambient(h, case["n"])
    original, original_ref = h.snapshot(cursor, reference, "retained_setup")
    midpoint = None
    names = ordinary_names(2)
    forks = 0
    previous_parent = None
    for cycle in range(case["cycles"]):
        value = h.tokens.atom("hot-overwrite-" + str(cycle))
        h.set(cursor, reference, names[0], value, cycle % 2 == 0)
        if cycle == case["cycles"] - 1:
            h.set(cursor, reference, names[0], value, False)
        if previous_parent is not None:
            h.get(previous_parent[0], previous_parent[1], names[0], "fork_isolation")
        previous_parent = (cursor, reference.fork())
        cursor, reference = h.fork(cursor, reference)
        forks += 1
        h.delete(cursor, reference, names[1])
        h.get(cursor, reference, names[1])
        h.get(previous_parent[0], previous_parent[1], names[1], "fork_isolation")
        previous_parent = (cursor, reference.fork())
        cursor, reference = h.fork(cursor, reference)
        forks += 1
        h.set(cursor, reference, names[1], h.tokens.atom("hot-reinsert-" + str(cycle)), True)
        h.get(previous_parent[0], previous_parent[1], names[1], "fork_isolation")
        previous_parent = (cursor, reference.fork())
        cursor, reference = h.fork(cursor, reference)
        forks += 1
        if cycle + 1 == case["cycles"] // 2:
            midpoint = h.snapshot(cursor, reference, "retained_setup")
    assert forks == 3 * case["cycles"]
    h.report["real_cursor_forks"] = forks
    h.ordered(cursor, reference)
    h.ordered(original, original_ref, prefix="historical", repeats=1)
    h.ordered(midpoint[0], midpoint[1], prefix="historical", repeats=1)
    final, final_ref = h.snapshot(cursor, reference, "join_setup")
    h.join([midpoint[0], final], [midpoint[1], final_ref])


def collision_case(h, case):
    names = collision_names(case["n"])
    fresh = tuple(CollidingName(str(names[i])) for i in (0, case["n"] // 2, case["n"] - 1))
    missing = CollidingName("c-missing")
    check_closed_domain((*names, *fresh, missing))
    cursor, reference = h.empty()
    parents = []
    for i, name in enumerate(names):
        h.set(cursor, reference, name, h.tokens.atom("collision-" + str(i)), i % 4 != 0)
        if parents:
            h.get(parents[-1][0], parents[-1][1], name, "fork_isolation")
        parents.append((cursor, reference.fork()))
        cursor, reference = h.fork(cursor, reference)
    for name in (*fresh, missing):
        h.get(cursor, reference, name, "collision_lookup")
    for cycle in range(case["hot_cycles"]):
        h.set(cursor, reference, CollidingName(str(names[0])),
              h.tokens.atom("collision-hot-" + str(cycle)), False)
        cursor, reference = h.fork(cursor, reference)
        h.delete(cursor, reference, CollidingName(str(names[1])))
        cursor, reference = h.fork(cursor, reference)
        h.set(cursor, reference, CollidingName(str(names[1])),
              h.tokens.atom("collision-reinsert-" + str(cycle)), True)
        cursor, reference = h.fork(cursor, reference)
    h.ordered(cursor, reference)
    reversed_ref = Reference()
    for name in reversed(names):
        reversed_ref.set(name, reference.values[name], False)
    bulk, bulk_ref = h.bulk(reversed_ref, "collision_bulk")
    h.ordered(bulk, bulk_ref, prefix="collision_bulk_first", repeats=2)
    snapshot, snapshot_ref = h.snapshot(cursor, reference, "join_setup")
    h.join([snapshot, bulk], [snapshot_ref, bulk_ref])
    for index in (0, case["n"] // 2 - 1, case["n"] - 1):
        h.ordered(parents[index][0], parents[index][1], prefix="historical", repeats=1)
    h.collision_evidence()


def bulk_reference(h, n, round_number):
    names = ordinary_names(n)
    rotation = round_number % n
    rotated = names[rotation:] + names[:rotation]
    left = dict.fromkeys(rotated)
    right = dict.fromkeys(tuple(reversed(rotated)))
    expected = Reference()
    for name in set().union(left.keys(), right.keys()):
        index = int(name[1:])
        value = h.tokens.atom("bulk-" + str(round_number // 2) + "-" + str(index))
        expected.set(name, value, round_number % 2 == 0 and index % 3 != 0)
    return expected


def invalid_bulk_controls(h):
    name = "u0000"
    entry = h.api.Entry(h.tokens.atom("invalid-boundary-control"), no_work=True)
    controls = (
        (CountedEntries(((name, entry), (name, entry))), ValueError, "duplicate"),
        (CountedEntries(((name, entry, "extra"),)), TypeError, "malformed-pair"),
    )
    for sequence, error_type, label in controls:
        before = sequence.entries
        try:
            h.ledger.call("invalid_boundary", label, lambda: h.api.NameVersion.from_unique_entries(
                h.ledger.meter, sequence))
        except h.api.BudgetExceeded:
            raise AssertionError("budget exhaustion is not malformed-input rejection")
        except error_type:
            pass
        else:
            raise AssertionError("invalid bulk input accepted: " + label)
        assert sequence.entries is before
        h.report.setdefault("invalid_bulk", []).append({
            "kind": label, "input_iterations": sequence.iterations,
            "input_visits": sequence.visits, "error": error_type.__name__})
    control = Reference({name: entry.value}, {name: True})
    version, expected = h.bulk(control, "invalid_boundary_continuation")
    h.ordered(version, expected, prefix="invalid_boundary_audit", repeats=1)


def bulk_case(h, case):
    versions, references = [], []
    for round_number in range(case["rebuilds"]):
        expected = bulk_reference(h, case["n"], round_number)
        version, expected = h.bulk(expected)
        versions.append(version)
        references.append(expected)
        h.ordered(version, expected, prefix="bulk_first", repeats=1)
    h.ordered(versions[-1], references[-1], prefix="bulk_final", repeats=2)
    for version, expected in zip(versions[:-1], references[:-1], strict=True):
        h.ordered(version, expected, prefix="historical", repeats=1)
    h.join(versions[2:4], references[2:4])
    assert set(h.report["callbacks"][-1]["required_names"]) == set(ordinary_names(case["n"]))
    if case["n"] == 32:
        invalid_bulk_controls(h)
        for version, expected in zip(versions, references, strict=True):
            h.ordered(version, expected, prefix="invalid_boundary_old_audit", repeats=1)


def retention_case(h, case):
    cursor, reference = ambient(h, case["ambient"])
    overwrite, deleted = "victim-overwrite", "victim-delete"
    weak = {}
    for name in (overwrite, deleted):
        sentinel = Value(("weak-sentinel", name))
        weak[name] = weakref.ref(sentinel)
        h.set(cursor, reference, name, sentinel, True, "retention_setup")
    del sentinel
    old = h.ledger.call("retention_setup", "snapshot-positive-owner", cursor.snapshot)
    assert isinstance(old, h.api.NameVersion)
    current = Value(("replacement-current",))
    h.set(cursor, reference, overwrite, current, True, "retention_mutation")
    h.delete(cursor, reference, deleted, "retention_mutation")
    newest = h.ledger.call("retention_publication", "snapshot-changed", cursor.snapshot)
    assert isinstance(newest, h.api.NameVersion)
    h.report["successful_changed_publications"] = 1
    h.get(cursor, reference, overwrite, "retention_current")
    h.get(newest, reference, overwrite, "retention_current")
    h.get(cursor, reference, deleted, "retention_current")
    h.get(newest, reference, deleted, "retention_current")
    gc.collect()
    for name in (overwrite, deleted):
        assert weak[name]() is not None, "retained old snapshot lost its value"
        assert h.ledger.call("retention_positive_owner", "get",
                            lambda: old.get(name)) is weak[name]()
    h.report["positive_old_owner"] = True
    del old
    gc.collect()
    released = {name: weak[name]() is None for name in (overwrite, deleted)}
    h.report["obsolete_values_released"] = released
    assert all(released.values()), "current root/path/history retains an obsolete value"
    h.ordered(newest, reference, prefix="retention_newest", repeats=2)
    h.set(cursor, reference, "after-retention", h.tokens.atom("after-retention"), False,
          "continuation_mutation")
    child, child_ref = h.fork(cursor, reference, "continuation_publication")
    h.ordered(child, child_ref, prefix="continuation", repeats=2)
    h.ordered(newest, Reference(
        {key: value for key, value in reference.values.items() if key != "after-retention"},
        {key: proof for key, proof in reference.no_work.items() if key != "after-retention"}),
        prefix="historical", repeats=1)


def atomic_prepare(h, case):
    ordinary = ordinary_names(case["ordinary_names"])
    colliding = collision_names(case["collision_names"] + 1)
    check_closed_domain((*ordinary, *colliding, CollidingName("c-missing")))
    reference = Reference()
    for i, name in enumerate((*ordinary, *colliding[:-1])):
        reference.set(name, h.tokens.atom("atomic-original-" + str(i)), True)
    version, reference = h.bulk(reference, "atomic_setup")
    cursor, reference = h.cursor(version, reference, "atomic_setup")
    old, old_ref = h.snapshot(cursor, reference, "atomic_setup")
    old_keys = h.ledger.call("atomic_setup", "keys", cursor.keys)
    old_items = h.ledger.call("atomic_setup", "ordered_items", cursor.ordered_items)
    checked_keys(old_keys, old_ref)
    checked_items(old_items, old_ref)
    h.set(cursor, reference, ordinary[0], h.tokens.atom("atomic-tail-u0"), False, "atomic_setup")
    h.delete(cursor, reference, ordinary[1], "atomic_setup")
    h.set(cursor, reference, ordinary[1], h.tokens.atom("atomic-tail-u1"), True, "atomic_setup")
    h.set(cursor, reference, colliding[0], h.tokens.atom("atomic-tail-c0"), False, "atomic_setup")
    h.set(cursor, reference, colliding[-1], h.tokens.atom("atomic-new-c16"), True, "atomic_setup")
    return cursor, reference, (old, old_ref, old_keys, old_items)


def event_signature(events):
    return [{key: event[key] for key in ("phase", "operation", "units", "counts")}
            for event in events]


def atomic_case(h, case, offset, baseline):
    cursor, reference, witnesses = atomic_prepare(h, case)
    target = case["target"]
    sequence = CountedEntries(tuple(
        (name, h.api.Entry(value, no_work=reference.no_work[name]))
        for name, value in reference.values.items()))
    original_entries = sequence.entries
    def perform(phase):
        if target == "from_unique_entries":
            return h.bulk_call(sequence, phase)
        return h.ledger.call(phase, target, getattr(cursor, target))
    if offset is not None:
        assert baseline is not None, "fault replay requires completed pristine baseline"
        calibration = baseline["calibration"]
        cost = calibration["units"]
        allowance = {"first": 0, "middle": cost // 2, "last": cost - 1}[offset]
        before = h.ledger.meter.used
        configured = before + allowance
        assert cost > 0 and 0 <= allowance < cost and configured <= MAXIMUM
        h.ledger.meter.limit = configured
        try:
            try:
                perform("failed_operation")
            except h.api.BudgetExceeded:
                spent = h.ledger.meter.used
                assert spent > configured and spent >= before
            else:
                raise AssertionError("configured target did not raise BudgetExceeded")
        finally:
            h.ledger.meter.limit = MAXIMUM
        failed_event = dict(h.ledger.events[-1])
        got = perform("retry_operation")
        measured = h.ledger.events[-1]
        assert measured["units"] == calibration["units"], "retry charged delta differs"
        assert measured["counts"] == calibration["counts"], "retry category delta differs"
        h.report["fault"] = {
            "offset": offset, "allowance": allowance, "used_before": before,
            "configured_limit": configured, "used_after_failure": spent,
            "failed_event": failed_event, "retry_units": measured["units"],
            "retry_counts": measured["counts"], "restored_limit": MAXIMUM,
            "validation_reads_excluded": True}
    else:
        got = perform("target_operation")
        measured = h.ledger.events[-1]
    assert sequence.entries is original_entries, "bulk supplier contents were replaced"
    assert measured["units"] > 0
    h.report["calibration"] = {"units": measured["units"], "counts": measured["counts"]}
    continuation_start = len(h.ledger.events)
    operation_view = None
    operation_parent = None
    if target in {"snapshot", "from_unique_entries"}:
        assert isinstance(got, h.api.NameVersion)
        h.ordered(got, reference, prefix="result_validation", repeats=1)
        cursor, reference = h.cursor(got, reference, "continuation_setup")
        immutable_result = (got, reference.fork())
    elif target == "fork":
        assert isinstance(got, h.api.NameCursor) and got is not cursor
        h.ordered(got, reference, prefix="result_validation", repeats=1)
        operation_parent = (cursor, reference.fork())
        cursor = got
        immutable_result = None
    elif target == "keys":
        checked_keys(got, reference)
        operation_view = ("keys", got, reference.fork())
        immutable_result = None
    else:
        checked_items(got, reference)
        operation_view = ("ordered_items", got, reference.fork())
        immutable_result = None
    child, child_ref = h.fork(cursor, reference, "continuation_publication")
    same = reference.values["u0000"]
    h.set(cursor, reference, "u0000", same, False, "continuation_mutation")
    h.delete(child, child_ref, "u0001", "continuation_mutation")
    h.set(child, child_ref, "u0001", h.tokens.atom("atomic-after-child"), False,
          "continuation_mutation")
    parent_final, parent_ref = h.snapshot(cursor, reference, "continuation_publication")
    child_final, child_final_ref = h.snapshot(child, child_ref, "continuation_publication")
    h.join([parent_final, child_final], [parent_ref, child_final_ref], "continuation_join")
    for state, expected in ((parent_final, parent_ref), (child_final, child_final_ref)):
        h.ordered(state, expected, prefix="continuation", repeats=2)
    old, old_ref, old_keys, old_items = witnesses
    checked_keys(old_keys, old_ref)
    checked_items(old_items, old_ref)
    h.ordered(old, old_ref, prefix="historical", repeats=1)
    if immutable_result is not None:
        h.ordered(*immutable_result, prefix="result_retained", repeats=1)
    if operation_parent is not None:
        h.ordered(*operation_parent, prefix="fork_parent_retained", repeats=1)
    if operation_view is not None:
        kind, view, expected = operation_view
        (checked_keys if kind == "keys" else checked_items)(view, expected)
    h.report["continuation_costs"] = event_signature(h.ledger.events[continuation_start:])
    h.report["continuation_observations"] = list(h.report["observations"])
    h.collision_evidence()
    if offset is not None:
        assert h.report["continuation_costs"] == baseline["continuation_costs"], (
            "post-retry continuation costs differ from pristine")
        assert h.report["continuation_observations"] == baseline["continuation_observations"], (
            "post-retry continuation outputs differ from pristine")
